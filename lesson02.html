Future 與 async／await

耗時的工作（讀檔、等感測器）會回傳 Future。用 await 等結果；沒有 await 的話，程式會先往下執行。

Future<String> loadName() async {
  await Future.delayed(const Duration(seconds: 1));
  return 'MoveGo';
}

Future<void> main() async {
  print('開始');
  final name = await loadName();
  print('載入 $name');
  loadName().then((v) => print('then 拿到 $v'));
  print('不等待，先印這行');
}
// 開始
// 載入 MoveGo
// 不等待，先印這行
// then 拿到 MoveGo
2

錯誤處理

on 指定例外型別，catch 取得例外物件，finally 不論成功失敗都會執行。讀使用者輸入、解析 JSON 一定要包。

int toSteps(String input) {
  try {
    return int.parse(input);
  } on FormatException catch (e) {
    print('不是數字：${e.source}');
    return 0;
  } finally {
    print('檢查完畢');
  }
}
// toSteps('120') → 印「檢查完畢」，回傳 120
// toSteps('abc') → 印「不是數字：abc」「檢查完畢」，回傳 0
3

Stream 資料流

Stream 是「一連串陸續到達的資料」，之後接加速度計就是 Stream。async* 搭配 yield 產生資料，await for 或 listen 接收。

Stream<int> ticker(int n) async* {
  for (var i = 1; i <= n; i++) {
    await Future.delayed(const Duration(milliseconds: 100));
    yield i;
  }
}

await for (final t in ticker(3)) {
  print('tick $t');           // tick 1、tick 2、tick 3
}
final sub = ticker(2).listen((t) => print('listen $t'));
await sub.cancel();           // 離開頁面時記得取消
4

jsonDecode 與 jsonEncode

jsonDecode 回傳的是 dynamic，要用 as 轉成正確型別；JSON 裡的清單要用 List<String>.from 轉換。

import 'dart:convert';

const text = '{"id": 1, "condition": 1000, "radial": ["0xFFFDBA74"]}';
final map = jsonDecode(text) as Map<String, dynamic>;
final id = map['id'] as int;
final colors = List<String>.from(map['radial'] as List);
print('$id ${colors.first}');  // 1 0xFFFDBA74

print(jsonEncode({'date': '2027.03.01', 'steps': 500}));
// {"date":"2027.03.01","steps":500}
5

fromJson 與 toJson

資料類別固定寫一對：factory fromJson 把 Map 變物件，toJson 把物件變 Map。jsonEncode 會自動呼叫 toJson。

class Record {
  final String date;
  final int steps;
  Record(this.date, this.steps);

  factory Record.fromJson(Map<String, dynamic> j) =>
      Record(j['date'] as String, j['steps'] as int);

  Map<String, dynamic> toJson() => {'date': date, 'steps': steps};
}

final r = Record.fromJson({'date': '2027.03.01', 'steps': 500});
print(jsonEncode([r, Record('2027.03.02', 800)]));
// [{"date":"2027.03.01","steps":500},{"date":"2027.03.02","steps":800}]
6

集合進階

sort 會直接改原清單；fold 放進 print 或字串時要寫 fold<int>，否則型別推斷會出錯。

final steps = [800, 300, 1200, 500];
steps.sort();                              // [300, 500, 800, 1200]
steps.sort((a, b) => b.compareTo(a));      // [1200, 800, 500, 300]
print(steps.fold<int>(0, (sum, s) => sum + s));  // 2800
print(steps.firstWhere((s) => s < 600));         // 500
print(steps.where((s) => s > 5000).firstOrNull); // null
print(steps.every((s) => s > 0));                // true

final map = {'Mon': 3000, 'Tue': 4500};
for (final e in map.entries) {
  print('${e.key}=${e.value}');            // Mon=3000、Tue=4500
}
