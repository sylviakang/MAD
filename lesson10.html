CustomPainter 兩個方法

paint 負責畫，shouldRepaint 決定資料變了要不要重畫。畫布左上角是 (0, 0)，往右 x 變大、往下 y 變大。

class RingPainter extends CustomPainter {
  RingPainter({required this.progress, required this.color,
      required this.track, this.stroke = 16});

  final double progress; // 0～1
  final Color color;
  final Color track;
  final double stroke;

  @override
  void paint(Canvas canvas, Size size) { /* 下一張卡 */ }

  @override
  bool shouldRepaint(RingPainter old) =>
      old.progress != progress || old.color != color;
}
2

畫底圈和進度弧

角度用弧度：-pi / 2 是 12 點鐘，2 * pi 是一整圈。半徑要扣掉半個線寬，否則線條外側會被切掉。

@override
void paint(Canvas canvas, Size size) {
  final center = size.center(Offset.zero);
  final radius = (min(size.width, size.height) - stroke) / 2;
  final paint = Paint()
    ..style = PaintingStyle.stroke
    ..strokeWidth = stroke
    ..strokeCap = StrokeCap.round;

  canvas.drawCircle(center, radius, paint..color = track);

  final sweep = 2 * pi * progress.clamp(0.0, 1.0);
  if (sweep > 0) {
    canvas.drawArc(
      Rect.fromCircle(center: center, radius: radius),
      -pi / 2, sweep, false, paint..color = color,
    );
  }
}

微縮進度條也自己畫

近 8 日每一格的小進度條同樣用 CustomPainter，下一堂才能跟圓環一起做動畫。圓頭會多出半個線寬，左右各內縮。

@override
void paint(Canvas canvas, Size size) {
  final y = size.height / 2;
  final r = size.height / 2;
  final paint = Paint()
    ..strokeWidth = size.height
    ..strokeCap = StrokeCap.round;
  canvas.drawLine(Offset(r, y), Offset(size.width - r, y),
      paint..color = track);
  final p = progress.clamp(0.0, 1.0);
  if (p > 0) {
    final end = r + (size.width - 2 * r) * p;
    canvas.drawLine(Offset(r, y), Offset(end, y), paint..color = color);
  }
}

卡路里與里程

第 1 堂看過 35 * 0.04 會變成 1.4000000000000001。先乘整數再除，結果就乾淨了。

// 每步 0.04 kcal = 4 / 100；每步 0.0006 km = 6 / 10000
double kcalOf(int steps) => steps * 4 / 100;
double kmOf(int steps) => steps * 6 / 10000;

String fmtKcal(int steps) => kcalOf(steps).toStringAsFixed(1); // 341.7
String fmtKm(int steps) => kmOf(steps).toStringAsFixed(2);     // 5.13

int percentOf(int steps, int goal) =>
    goal <= 0 ? 0 : (steps * 100 / goal).round();

int get goal => _goal;

Future<void> setGoal(int goal) async {
  if (goal <= 0) return;
  _goal = goal;
  await _settings.writeAsString(jsonEncode({'goal': goal}), flush: true);
}

// 首頁
Future<void> _editGoal() async {
  final goal = await showDialog<int>(
    context: context,
    builder: (_) => GoalDialog(goal: repo.goal),
  );
  if (goal == null) return;
  await repo.setGoal(goal);
  setState(() {});
}
