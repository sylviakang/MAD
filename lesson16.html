Rect plotRect(Size size, {double bottom = 36}) =>
    Rect.fromLTRB(chartLeft, chartTop, size.width, size.height - bottom);

double yIn(Rect plot, double v, double yMax) =>
    plot.bottom - plot.height * (v / yMax);

// 第 i 個點（共 n 個）取格子中心
double xIn(Rect plot, int i, int n) =>
    plot.left + plot.width / n * (i + 0.5);

3

每個點的座標

X 軸平均分成 n 格，點放在格子中心；Y 用同一個公式。超過 yMax 的值先 clamp，線才不會畫出框外。

final plot = plotRect(size, bottom: 24);   // X 軸只有一行字
drawYGrid(canvas, plot, yMax, tick);

final n = values.length;
final points = [
  for (var i = 0; i < n; i++)
    Offset(xIn(plot, i, n),
        yIn(plot, values[i].clamp(0.0, yMax), yMax)),
];
4

Path 連成折線

第一點 moveTo，其餘 lineTo。Paint 預設是填滿，忘了設定 stroke 會畫出一塊奇怪的多邊形。

final path = Path()..moveTo(points.first.dx, points.first.dy);
for (final p in points.skip(1)) {
  path.lineTo(p.dx, p.dy);
}
canvas.drawPath(
  path,
  Paint()
    ..color = AppColors.primary
    ..style = PaintingStyle.stroke     // 一定要
    ..strokeWidth = 3
    ..strokeJoin = StrokeJoin.round,
);
