class TrendData {
  TrendData({required this.title, required this.labels,
      required this.values, required this.totalKcal,
      required this.dailyAvg});

  final String title;          // 每週平均／每月平均
  final List<String> labels;   // X 軸：Mon…Sun 或 Sep…Mar
  final List<double> values;   // 每個點的卡路里
  final double totalKcal;      // 總消耗卡路里
  final double dailyAvg;       // 每日平均
}

週的彙總

每天一個點（當天的卡路里）。總卡路里用第 7 堂作業的 stepsBetween 一次算完。

TrendData weekTrend(StepRepository repo, DateTime today) {
  final days = weekDays(today);
  final values = [for (final d in days) kcalOf(repo.stepsOfDay(d))];
  final total = kcalOf(repo.stepsBetween(days.first, days.last));
  final n = _daysPassed(days.first, days.last, today);
  return TrendData(
    title: '每週平均',
    labels: [for (final d in days) fmtE(d)],
    values: values,
    totalKcal: total,
    dailyAvg: n == 0 ? 0 : total / n,
  );
}
5

月的彙總

每個月一個點。圖表標題是「每月平均」，所以每個點放該月的每日平均，七個月才能互相比較。

for (final m in months) {
  final last = DateTime(m.year, m.month + 1, 0);
  final kcal = kcalOf(repo.stepsBetween(m, last));
  final n = _daysPassed(m, last, today);   // 本月只算到今天
  values.add(n == 0 ? 0 : kcal / n);
  total += kcal;
  totalDays += n;
}
// dailyAvg = total / totalDays
6

題目沒寫清楚怎麼辦

「每日平均」要除以幾天，題目沒有說。選一種合理的解讀，寫在註解裡，能說出理由，並且週、月用同一套規則。

// 解讀 A（本專案採用）：除以「已經過的天數」
//   今天週三 → 本週除以 3；未來的日子還沒走，不該拉低平均
// 解讀 B：一律除以 7（或該月天數）
// 解讀 C：只除以「有紀錄的天數」

// 未來的日子不算
int _daysPassed(DateTime from, DateTime to, DateTime today) {
  final t = dateOnly(today);
  final end = to.isAfter(t) ? t : to;
  if (end.isBefore(from)) return 0;
  final a = DateTime.utc(from.year, from.month, from.day);
  final b = DateTime.utc(end.year, end.month, end.day);
  return b.difference(a).inDays + 1;
}
