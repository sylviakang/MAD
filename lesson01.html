1

變數與型別

var 可以重新指定，final 就是 Kotlin 的 val，const 是編譯期常數。

void variables() {
  var steps = 1200;          // 型別推斷為 int
  final goal = 6000;         // 之後不能改
  const kcalPerStep = 0.04;  // 編譯期常數
  String title = 'MoveGo';
  steps += 300;
  print('$title：$steps / $goal 步，${steps * kcalPerStep} kcal');
  // MoveGo：1500 / 6000 步，60.0 kcal
  print(7 / 2);   // 3.5
  print(7 ~/ 2);  // 3
  print(3.14159.toStringAsFixed(2)); // 3.14
  print('5'.padLeft(2, '0'));        // 05
  print(int.parse('42') + 1);        // 43
}
2

null safety

型別後加 ? 才能放 null。從 Map 取值、從檔案讀資料都可能拿到 null，這是 MoveGo 最常見的當機原因。

String? findName(bool ok) => ok ? 'SK' : null;

void nullSafety() {
  int? saved;
  print(saved ?? 0);             // 0
  String? name = findName(false);
  print(name?.length);           // null
  saved ??= 100;                 // 是 null 才指定
  print(saved);                  // 100
}
3

函式

一行的函式用 =>。具名參數用 { }，Flutter 的 Widget 全都是這種寫法，一定要熟。

int add(int a, int b) => a + b;
String greet(String name, {String prefix = 'Hi', required int times}) =>
    List.filled(times, '$prefix $name').join(', ');
String label(int steps, [String unit = '步']) => '$steps $unit';

// add(1, 2)              → 3
// greet('SK', times: 2)  → Hi SK, Hi SK
// label(10)              → 10 步

5

類別

this.x 直接在建構子指定欄位；get 寫計算屬性；覆寫 toString 方便除錯。

class Badge {
  final String title;
  final int target;
  int current;
  Badge(this.title, this.target, {this.current = 0});
  Badge.locked(String title, int target) : this(title, target);
  bool get unlocked => current >= target;
  @override
  String toString() =>
      '$title（$current/$target）${unlocked ? '已解鎖' : '未解鎖'}';
}
// Badge('初次一萬步', 10000, current: 12000) → 初次一萬步（12000/10000）已解鎖
// Badge.locked('十萬步', 100000)            → 十萬步（0/100000）未解鎖
6

List 與 Map

where、map 回傳的是 Iterable，記得加 .toList()。... 與 collection-if 在 Flutter 組畫面時很常用。

final list = <int>[300, 1200, 800];
list.add(500);
final big = list.where((s) => s >= 500).toList();  // [1200, 800, 500]
final kcal = list.map((s) => s * 0.04).toList();  // [12.0, 48.0, 32.0, 20.0]
final more = [...list, 0, if (showTotal) 9999];   // [300, 1200, 800, 500, 0, 9999]

final map = <String, int>{'Mon': 3000};
map['Tue'] = 4500;
map.update('Mon', (v) => v + 100, ifAbsent: () => 0); // Mon → 3100
print(map.containsKey('Wed'));                         // false
