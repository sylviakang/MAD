ClipRRect：整條變圓角

進度條是兩個 Container 拼起來的，各自設圓角會變成四個圓角。改用 ClipRRect 把整個 Row 包起來，它會把裡面的東西「裁」成圓角。

import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(home: HomePage());
  }
}

class HomePage extends StatelessWidget {
  const HomePage({super.key});

  @override
  Widget build(BuildContext context) {
    final steps = 8234;
    final goal = 10000;

    return Scaffold(
      backgroundColor: Color(0xFFF4F7FB),
      appBar: AppBar(title: Text('進度條')),
      body: Padding(
        padding: EdgeInsets.all(16),
        ⟦child: ClipRRect(⟧
          ⟦borderRadius: BorderRadius.circular(8),⟧
          child: Row(
            children: [
              Expanded(flex: steps, child: Container(height: 16, color: Color(0xFF00A396))),
              Expanded(flex: goal - steps, child: Container(height: 16, color: Color(0xFFE0E6EE))),
            ],
          ),
        ⟦),⟧
      ),
    );
  }
}
