void printSummary(List<StepRecord> records) {
  var total = 0;
  var okDays = 0;
  var best = records[0];
  for (final r in records) {
    total += r.steps;
    if (r.reached) {
      okDays++;
    }
    if (r.steps > best.steps) {
      best = r;
    }
  }
  final avg = total / records.length;
  print('總步數：$total');
  print('平均：${avg.toStringAsFixed(1)}');
  print('達標：$okDays 天');
  print('最佳：$best');
}
