switch 運算式：直接得到一個值

  • 寫在等號右邊，每一行是 值 => 結果,
  • _ 代表「其他全部」，一定要有，不然 Dart 會說沒有涵蓋所有情況。
  • 最後的大括號後面要加分號。

完成 BMI 判斷器

  1. 沿用上一堂的 if / else if 算出等級。
  2. 用 switch 運算式依等級選建議。
  3. 理想體重＝18.5×身高² 到 24×身高²。
void main() {
  final heightCm = 172.0;
  ⟦final weightKg = 80.0;⟧
  final h = heightCm / 100;
  final bmi = weightKg / (h * h);
  String level;
  if (bmi < 18.5) {
    level = '過輕';
  } else if (bmi < 24) {
    level = '正常';
  } else if (bmi < 27) {
    level = '過重';
  } else {
    level = '肥胖';
  }
  ⟦final advice = switch (level) {⟧
  ⟦  '過輕' => '多吃一點，加強肌力訓練',⟧
  ⟦  '正常' => '繼續保持！',⟧
  ⟦  '過重' => '每天多走 2000 步',⟧
  ⟦  _ => '建議諮詢醫師或營養師',⟧
  ⟦};⟧
  ⟦final minW = 18.5 * h * h;⟧
  ⟦final maxW = 24 * h * h;⟧

  print('===== BMI 判斷器 =====');
  print('身高：$heightCm cm　體重：$weightKg kg');
  print('BMI ：${bmi.toStringAsFixed(1)}（$level）');
  ⟦print('理想體重：${minW.toStringAsFixed(1)} ～ ${maxW.toStringAsFixed(1)} kg');⟧
  ⟦print('建議：$advice');⟧
}
