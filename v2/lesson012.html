箭頭函式：一行就寫完

命名參數和預設值

  • 參數用 { } 包起來就變成命名參數，呼叫時要寫名字：caloriesOf(8000, perStep: 0.05)。
  • double perStep = 0.04：沒給的時候用預設值。
  • required：一定要給，忘了 Dart 會提醒。
void main() {
  ⟦print(caloriesOf(8000));⟧
  ⟦print(caloriesOf(8000, perStep: 0.05));⟧
  ⟦printTable(size: 3);⟧

完成單元 4：乘法表＋一週卡路里報表

  1. printTable 改成有預設值 9，不給就印九九乘法表。
  2. 寫一個 report 函式，用 for 迴圈印出一週 7 天的步數、卡路里、公里數。

void main() {
  ⟦printTable(size: 5);⟧
  ⟦print('');⟧
  ⟦report(startSteps: 6000, addPerDay: 800);⟧

double caloriesOf(int steps, {double perStep = 0.04}) =>
    steps * perStep;
double kmOf(int steps) => steps * 0.0007;

void printTable({⟦int size = 9⟧}) {
  for (var a = 1; a <= size; a++) {
    var line = '';
    for (var b = 1; b <= size; b++) {
      line += '${a}x$b=${(a * b).toString().padLeft(2)} ';
    }
    print(line);
  }
}

⟦void report({required int startSteps, int addPerDay = 0}) {⟧
⟦  print('天   步數   大卡  公里');⟧
⟦  var steps = startSteps;⟧
⟦  for (var day = 1; day <= 7; day++) {⟧
⟦    final s = steps.toString().padLeft(5);⟧
⟦    final kcal = caloriesOf(steps).toStringAsFixed(1).padLeft(6);⟧
⟦    final km = kmOf(steps).toStringAsFixed(2).padLeft(5);⟧
⟦    print(' $day  $s $kcal $km');⟧
⟦    steps += addPerDay;⟧
⟦  }⟧
⟦}⟧
