萬用的 SensorEffect

第 9 堂的陀螺儀寫了一整段 DisposableEffect。把它變成可以重複用的函式，換感測器只要換 type。rememberUpdatedState 讓監聽器永遠呼叫最新的 lambda。

@Composable
fun SensorEffect(type: Int, onValues: (FloatArray, Long) -> Unit) {
    val context = LocalContext.current
    val latest by rememberUpdatedState(onValues)
    DisposableEffect(type) {
        val sm = context.getSystemService(Context.SENSOR_SERVICE) as SensorManager
        val sensor = sm.getDefaultSensor(type)
        val listener = object : SensorEventListener {
            override fun onSensorChanged(e: SensorEvent) { latest(e.values, e.timestamp) }
            override fun onAccuracyChanged(sensor: Sensor?, accuracy: Int) {}
        }
        if (sensor != null) sm.registerListener(listener, sensor, SensorManager.SENSOR_DELAY_GAME)
        onDispose { sm.unregisterListener(listener) }
    }
}
2

加速度計量的是重力方向

手機平放時 z 約 9.81；往右傾，重力跑到 x 軸變成負的。所以「傾斜多少」直接看數值，不用像陀螺儀那樣積分，也不會越飄越遠。

// 右邊往下傾：ax 變負 → 游標往右
fun accelToX(center: Float, ax: Float, rangePx: Float): Float =
    center - ax / G * rangePx
3

低通濾波去抖動

加速度計的值會一直小幅跳動，游標看起來在發抖。每次只往新值移動 20%，就會變平滑。搖一搖偵測則要用原始值。

fun lowPass(prev: Float, value: Float, alpha: Float = 0.2f): Float =
    prev + alpha * (value - prev)

ax = lowPass(ax, v[0])
4

「持續一段時間」與「冷卻時間」

傾斜選答要同一方向維持 1 秒才確認；搖一搖觸發後 1 秒內不再算。兩者都是記下時間，跟現在時間比較，跟第 11 堂停留 3 秒同一個想法。

class ShakeDetector(private val thresholdG: Float = 2.5f, private val cooldownMs: Long = 1000) {
    private var lastShake = -cooldownMs
    fun onSample(x: Float, y: Float, z: Float, nowMs: Long): Boolean {
        if (magnitude(x, y, z) / G < thresholdG) return false
        if (nowMs - lastShake < cooldownMs) return false
        lastShake = nowMs
        return true
    }
}

SensorMath.kt
const val G = 9.81f

// 低通濾波：新值只取一部分，數值不會抖動。alpha 越小越平滑、越慢
fun lowPass(prev: Float, value: Float, alpha: Float = 0.2f): Float =
    prev + alpha * (value - prev)

// 加速度計換成游標位置（直向拿手機）
// 右邊往下傾：ax 變負 → 游標往右；上方往下傾：ay 變小 → 游標往上
fun accelToX(center: Float, ax: Float, rangePx: Float): Float = center - ax / G * rangePx
fun accelToY(center: Float, ay: Float, baseAy: Float, rangePx: Float): Float =
    center + (ay - baseAy) / G * rangePx

// 三軸合力
fun magnitude(x: Float, y: Float, z: Float): Float = sqrt(x * x + y * y + z * z)

// 傾斜方向：超過門檻才算，取傾斜比較多的那一軸
enum class Tilt { NONE, LEFT, RIGHT, UP, DOWN }

fun tiltDirection(ax: Float, ay: Float, baseAy: Float, threshold: Float = 3f): Tilt {
    val dx = -ax
    val dy = ay - baseAy
    if (abs(dx) < threshold && abs(dy) < threshold) return Tilt.NONE
    return if (abs(dx) >= abs(dy)) {
        if (dx > 0) Tilt.RIGHT else Tilt.LEFT
    } else {
        if (dy > 0) Tilt.DOWN else Tilt.UP
    }
}

// 搖一搖：合力超過 2.5 倍重力算一次，之後 1 秒內不再算
class ShakeDetector(private val thresholdG: Float = 2.5f, private val cooldownMs: Long = 1000) {
    private var lastShake = -cooldownMs
    fun onSample(x: Float, y: Float, z: Float, nowMs: Long): Boolean {
        if (magnitude(x, y, z) / G < thresholdG) return false
        if (nowMs - lastShake < cooldownMs) return false
        lastShake = nowMs
        return true
    }
}

// 同一個方向持續 holdMs 才確認（傾斜選答用）
class HoldDetector(private val holdMs: Long = 1000) {
    private var current = Tilt.NONE
    private var since = 0L
    fun onSample(t: Tilt, nowMs: Long): Tilt {
        if (t != current) {
            current = t
            since = nowMs
            return Tilt.NONE
        }
        return if (t != Tilt.NONE && nowMs - since >= holdMs) t else Tilt.NONE
    }
    fun reset() { current = Tilt.NONE }
}

// 光線分級
fun lightLevel(lux: Float): String = when {
    lux < 10f -> "暗"
    lux < 1000f -> "室內"
    else -> "戶外"
}

06

旋轉向量：指南針

getRotationMatrixFromVector＋getOrientation

方位角 0° 是北方，順時針增加。getOrientation 回傳 -180～180，加 360 再取餘數變成 0～360：

SensorLab.kt
    // 6. 旋轉向量 → 方位角（指南針）
    var azimuth by remember { mutableFloatStateOf(0f) }
    SensorEffect(Sensor.TYPE_ROTATION_VECTOR) { v, _ ->
        val r = FloatArray(9)
        val o = FloatArray(3)
        SensorManager.getRotationMatrixFromVector(r, v)
        SensorManager.getOrientation(r, o)
        azimuth = (Math.toDegrees(o[0].toDouble()).toFloat() + 360f) % 360f
    }
完成後轉動手機，方位角在 0～360 之間變化。
