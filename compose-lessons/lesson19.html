四個重點概念

講解 20 分鐘

1

刪除要靠 id

原本 getAll 沒有讀 id，所以畫面上不知道要刪哪一筆。GameResult 加一個有預設值的 id，剛玩完還沒存的成績就是 0，其他地方都不用改。

data class GameResult(
    val correct: Int, val wrong: Int,
    val durationSec: Int, val endMillis: Long,
    val id: Long = 0
)
2

? 帶值，不要自己接字串

條件的值用 ? 佔位，再用陣列帶進去。排序欄位不能用 ?，所以用 enum 限定只有幾種寫法，避免打錯字或被亂塞。

db.delete("records", "id = ?", arrayOf(id.toString()))

enum class SortBy(val sql: String, val label: String) {
    NEWEST("end_time DESC", "最新"),
    BEST("correct DESC, duration ASC", "最佳")
}
3

改了資料庫，畫面要重新讀

remember 的 key 放一個版本號，刪除後 version++，remember 就會重新查資料庫。排序改變也一樣放進 key。

var version by remember { mutableIntStateOf(0) }
val records = remember(sort, version) { db.getAll(sort) }

db.delete(target.id); version++
4

JSON 欄位不一定存在

getString 找不到欄位會閃退；optString、optInt 找不到時回傳你給的預設值。題本說「可能沒有」的欄位一律用 opt。

obj.optString("category", "一般")
obj.optInt("score", 1)

// 排序方式：只允許這幾種，避免把任意字串接進 SQL
enum class SortBy(val sql: String, val label: String) {
    NEWEST("end_time DESC", "最新"),
    BEST("correct DESC, duration ASC", "最佳"),
    FASTEST("duration ASC", "最快")
}
RecordDb.kt
    // 多讀 id，刪除時要用
    fun getAll(sort: SortBy = SortBy.NEWEST): List<GameResult> {
        val list = mutableListOf<GameResult>()
        val c = readableDatabase.rawQuery(
            "SELECT id, correct, wrong, duration, end_time FROM records ORDER BY ${sort.sql}", null
        )
        while (c.moveToNext()) {
            list.add(GameResult(c.getInt(1), c.getInt(2), c.getInt(3), c.getLong(4), id = c.getLong(0)))
        }
        c.close()
        return list
    }

刪除、清除、統計、最佳

RecordDb 加 4 個函式
RecordDb.kt（class 外面）
data class Stats(val count: Int, val avgCorrect: Double, val maxCorrect: Int)
RecordDb.kt
    fun delete(id: Long) {
        writableDatabase.delete("records", "id = ?", arrayOf(id.toString()))
    }

    fun clearAll() {
        writableDatabase.delete("records", null, null)
    }

    // 聚合函數：一次算出筆數、平均、最高
    fun stats(): Stats {
        val c = readableDatabase.rawQuery(
            "SELECT COUNT(*), IFNULL(AVG(correct), 0), IFNULL(MAX(correct), 0) FROM records", null
        )
        c.moveToFirst()
        val s = Stats(c.getInt(0), c.getDouble(1), c.getInt(2))
        c.close()
        return s
    }

    // 最佳一筆：答對最多，同分比時間短；沒有紀錄回傳 null
    fun best(): GameResult? = getAll(SortBy.BEST).firstOrNull()

// 變化 1：options 是陣列、answer 是數字 0～3
// [{"qid":"q1","question":"…","options":["A","B","C","D"],"answer":2}]
fun parseArrayOptions(text: String): List<Question> {
    val array = JSONArray(text)
    val list = mutableListOf<Question>()
    for (i in 0 until array.length()) {
        val obj = array.getJSONObject(i)
        val opts = obj.getJSONArray("options")
        list.add(
            Question(
                qid = obj.getString("qid"),
                question = obj.getString("question"),
                options = (0 until opts.length()).map { opts.getString(it) },
                answer = "abcd"[obj.getInt("answer")].toString()
            )
        )
    }
    return list
}
JsonVariants.kt
// 變化 2：外面多包一層，題目在 "data" 陣列裡
// {"version":2,"data":[{"qid":…,"options":{"a":…},"answer":"b"}]}
fun parseWrapped(text: String): List<Question> {
    val data = JSONObject(text).getJSONArray("data")
    return parseObjectOptions(data)
}

JsonVariants.kt
// 變化 3：有些欄位可能沒有 → 用 opt 系列給預設值，不會閃退
// category 沒有時算 "一般"，score 沒有時算 1
data class QuizMeta(val qid: String, val category: String, val score: Int)

fun parseMeta(text: String): List<QuizMeta> {
    val array = JSONArray(text)
    return (0 until array.length()).map {
        val obj = array.getJSONObject(it)
        QuizMeta(obj.getString("qid"), obj.optString("category", "一般"), obj.optInt("score", 1))
    }
}

// 變化 4：依分類分組，數每一類有幾題
fun countByCategory(metas: List<QuizMeta>): Map<String, Int> =
    metas.groupBy { it.category }.mapValues { it.value.size }
